$$
z=\sqrt{ 3x+\tan^2(y) }
$$
$$
x=\frac{u^2}{v}
$$
$$
y=3u^2v
$$
$$
\frac{dz}{dv}=\frac{dz}{dx} \frac{dx}{dv}+\frac{dz}{dy} \frac{dy}{dv}
$$
$$
\frac{dz}{du}=\frac{dz}{dx} \frac{dx}{du}+\frac{dz}{dy} \frac{dy}{du}
$$
$$
\frac{dz}{dx}=\frac{1}{2}\cdot(3x+\tan^2(y))^{-1/2} \cdot 3=\frac{3}{2(3x+\tan^2(y))^{1/2}}
$$
$$
\frac{dz}{dy}=\frac{1}{2}\cdot(3x+\tan^2(y))^{-1/2}\cdot2\cdot \frac{1}{\cos^2(y)}\cdot1=\frac{1}{\cos^2(y)\cdot(3x+\tan^2(y))}
$$
$$
\frac{dx}{dv}=(u^2v^{-1})'=(-1)u^2(v^{-2})=- \frac{u^2}{v^2}
$$
$$
\frac{dy}{dv}=3u^2
$$
$$
\frac{dx}{du}=2\cdot \frac{1}{v}\cdot u=\frac{2u}{v}
$$
$$
\frac{dy}{du}=3\cdot2\cdot v\cdot u=6vu
$$
$$\frac{dz}{dv}=\frac{3}{2(3x+\tan^2(y))^{1/2}}\cdot \left( - \frac{u^2}{v^2} \right)  +\frac{1}{\cos^2(y)\cdot(3x+\tan^2(y))} \cdot 3u^2=$$
$$
=\frac{3u^2}{\cos^2(y)\cdot(3x+\tan^2(y))^{1/2}}-\frac{3u^2}{2v^2(3x+\tan^2(y))}=
$$
$$
=\frac{3u^2}{\cos^2(3u^2v)\cdot(3\frac{u^2}{v}+\tan^2(3u^2v))^{1/2}}-\frac{3u^2}{2v^2(3 \frac{u^2}{v}+\tan^2(3u^2v))}
$$
$$
\frac{dz}{du}=\frac{3}{2(3x+\tan^2(y))^{1/2}} \cdot \frac{2u}{v} + \frac{1}{\cos^2(y)\cdot(3x+\tan^2(y))} \cdot 6vu=
$$
$$
=\frac{6u}{2v(3x+\tan^2(y))^{1/2}}+\frac{6vu}{\cos^2(y)(3x+\tan^2(y))}=
$$
$$
=\frac{6u}{2v(3 \frac{u^2}{v}+\tan^2(3u^2v))^{1/2}}+\frac{6vu}{\cos^2(3u^2v)(3\frac{u^2}{v}+\tan^2(3u^2v))}
$$
